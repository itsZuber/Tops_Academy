Question:
Build an array called cart containing 4 objects, each representing a Flipkart product with properties: productName, price, and quantity. Write a loop to display each product's name and total price (price × quantity) in the console.

Answer:

let cart = [
    {
        productName: "Laptop",
        price: 50000,
        quantity: 1
    },
    {
        productName: "Mouse",
        price: 800,
        quantity: 2
    },
    {
        productName: "Keyboard",
        price: 1500,
        quantity: 1
    },
    {
        productName: "Headphones",
        price: 2000,
        quantity: 2
    }
];

for (let product of cart) {
    let totalPrice = product.price * product.quantity;

    console.log(product.productName + " - Total Price: " + totalPrice);
}

Question:
Create an array called orders, where each order is an object with properties: orderId, items (an array of objects with itemName and price), and delivery (an object with address and status). Write a loop to display each order's orderId and total price of all items.

Answer:

let orders = [
    {
        orderId: 101,
        items: [
            {
                itemName: "Pizza",
                price: 300
            },
            {
                itemName: "Burger",
                price: 200
            }
        ],
        delivery: {
            address: "Ahmedabad",
            status: "Delivered"
        }
    },
    {
        orderId: 102,
        items: [
            {
                itemName: "Biryani",
                price: 250
            },
            {
                itemName: "Cold Drink",
                price: 80
            }
        ],
        delivery: {
            address: "Ahmedabad",
            status: "Out for Delivery"
        }
    }
];

for (let order of orders) {
    let totalPrice = 0;

    for (let item of order.items) {
        totalPrice = totalPrice + item.price;
    }

    console.log(order.orderId + " - Total Price: " + totalPrice);
}
