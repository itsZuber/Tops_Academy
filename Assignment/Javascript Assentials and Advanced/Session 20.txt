Bug:
When I clicked the Add to Cart button multiple times, the same product was added more than once.

Fix:
I added a check before adding the product to the cart. If the product already exists in the cart, it will not be added again.

function addToCart(product) {
    let alreadyAdded = cart.some(function(item) {
        return item.id === product.id;
    });

    if (!alreadyAdded) {
        setCart([...cart, product]);
    }
}

Question:
Create a login form with email and password fields and add basic UI validation: show an error message if either field is left empty when the user clicks the Login button.

Answer:

import { useState } from "react";

function Login() {
    let [email, setEmail] = useState("");
    let [password, setPassword] = useState("");
    let [emailError, setEmailError] = useState("");
    let [passwordError, setPasswordError] = useState("");

    function handleLogin(event) {
        event.preventDefault();

        setEmailError("");
        setPasswordError("");

        if (email === "") {
            setEmailError("Email is required");
        }

        if (password === "") {
            setPasswordError("Password is required");
        }

        if (email !== "" && password !== "") {
            alert("Login successful");
        }
    }

    return (
        <form onSubmit={handleLogin}>
            <input
                type="email"
                placeholder="Email"
                value={email}
                onChange={(event) => setEmail(event.target.value)}
            />

            {emailError && <p>{emailError}</p>}

            <br />

            <input
                type="password"
                placeholder="Password"
                value={password}
                onChange={(event) => setPassword(event.target.value)}
            />

            {passwordError && <p>{passwordError}</p>}

            <br />

            <button type="submit">Login</button>
        </form>
    );
}

export default Login;

Question:
Test a Zomato-style review form in your app: fill out all fields, submit, then try submitting with missing or invalid data (like empty review or rating above 5). List any bugs or missing validation you find and fix at least one.

Answer:

Testing Result:

Filled all fields and submitted → Review was submitted successfully.
Left the review empty → Form should show an error.
Entered rating above 5 → Validation was missing.
Entered rating below 1 → Validation was missing.

Bug Found:
The rating field allowed values greater than 5.

Fix:

import { useState } from "react";

function ReviewForm() {
    let [review, setReview] = useState("");
    let [rating, setRating] = useState("");
    let [error, setError] = useState("");

    function handleSubmit(event) {
        event.preventDefault();

        if (review.trim() === "") {
            setError("Review is required");
            return;
        }

        if (rating < 1 || rating > 5) {
            setError("Rating must be between 1 and 5");
            return;
        }

        setError("");
        alert("Review submitted");
    }

    return (
        <form onSubmit={handleSubmit}>
            <textarea
                placeholder="Write your review"
                value={review}
                onChange={(event) => setReview(event.target.value)}
            ></textarea>

            <br /><br />

            <input
                type="number"
                placeholder="Rating 1-5"
                value={rating}
                onChange={(event) => setRating(event.target.value)}
            />

            {error && <p>{error}</p>}

            <br />

            <button type="submit">Add Review</button>
        </form>
    );
}

export default ReviewForm;
