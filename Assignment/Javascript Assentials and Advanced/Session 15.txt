Question:
Create a class called FlipkartProduct with properties: title, price, and rating. Write a method getDiscountedPrice(discountPercent) that returns the price after applying the given discount percentage.

Answer:

class FlipkartProduct {
    constructor(title, price, rating) {
        this.title = title;
        this.price = price;
        this.rating = rating;
    }

    getDiscountedPrice(discountPercent) {
        return this.price - (this.price * discountPercent / 100);
    }
}

let product = new FlipkartProduct("Redmi Note 12 Pro", 18999, 4.5);

console.log(product.getDiscountedPrice(10));

Question:
Extend the Playlist class by adding a method addSong(songName) that adds a new song to the songs array. Demonstrate by adding two songs and displaying the updated playlist.

Answer:

class Playlist {
    constructor(name, songs) {
        this.name = name;
        this.songs = songs;
    }

    addSong(songName) {
        this.songs.push(songName);
    }

    displayPlaylist() {
        console.log("Playlist: " + this.name);
        console.log("Songs: " + this.songs.join(", "));
    }
}

let playlist = new Playlist("Chill Vibes", [
    "Perfect",
    "Love Story",
    "Night Changes"
]);

playlist.addSong("Believer");
playlist.addSong("Faded");

playlist.displayPlaylist();
