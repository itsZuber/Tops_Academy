Question:
Given an array of Flipkart product objects with properties name, price, and isInStock, use the filter() method to create a new array containing only the products that are in stock.

Answer:

let products = [
    {
        name: "Laptop",
        price: 50000,
        isInStock: true
    },
    {
        name: "Mouse",
        price: 800,
        isInStock: false
    },
    {
        name: "Keyboard",
        price: 1500,
        isInStock: true
    },
    {
        name: "Headphones",
        price: 2000,
        isInStock: false
    }
];

let inStockProducts = products.filter(function(product) {
    return product.isInStock == true;
});

console.log(inStockProducts);

Question:
Given an array of Zomato order objects with a clicks property, use reduce() to calculate the total number of clicks across all orders.

    },
    {
        clicks: 250
    },

Question:
