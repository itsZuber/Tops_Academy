Question:
Build a small form with an input box and a button labeled 'Add Song'. When the button is clicked, use querySelector to get the input value and dynamically add a new <li> element with the song name to a <ul id="playlist"> on the page.

Answer:

<!DOCTYPE html>

<html lang="en"> <head> <meta charset="UTF-8"> <meta name="viewport" content="width=device-width, initial-scale=1.0"> <title>Playlist</title> </head> <body>

<input type="text" id="song" placeholder="Enter song name">
<button id="addSong">Add Song</button>

<ul id="playlist"></ul>

<script>
    document.querySelector("#addSong").addEventListener("click", function() {
        let songName = document.querySelector("#song").value;

        let li = document.createElement("li");
        li.innerHTML = songName;

        document.querySelector("#playlist").appendChild(li);
    });
</script>

</body> </html>

Question:
Create a 'Dark Mode' toggle button that, when clicked, uses getElementById to change the background color of the page and updates the button text to 'Light Mode'.

Hint: Use the style property to change document.body's background color.

Answer:

<!DOCTYPE html>

<html lang="en"> <head> <meta charset="UTF-8"> <meta name="viewport" content="width=device-width, initial-scale=1.0"> <title>Dark Mode</title> </head> <body>

<button id="modeButton">Dark Mode</button>

<script>
    let modeButton = document.getElementById("modeButton");

    modeButton.addEventListener("click", function() {
        document.body.style.backgroundColor = "black";
        document.body.style.color = "white";

        modeButton.innerHTML = "Light Mode";
    });
</script>

</body> </html>




Question:
Create a function that updates the background color of a <div id="moodbox"></div> in real-time as the user types a color name (like 'pink' or 'yellow') into an input box.

Hint: Listen for the input event and use .style.backgroundColor.

Answer:

<!DOCTYPE html>

<html lang="en"> <head> <meta charset="UTF-8"> <meta name="viewport" content="width=device-width, initial-scale=1.0"> <title>Mood Box</title> </head> <body>

<input type="text" id="colorInput" placeholder="Enter a color">

<div id="moodbox" style="width: 200px; height: 100px; margin-top: 20px;"></div>

<script>
    let colorInput = document.getElementById("colorInput");
    let moodbox = document.getElementById("moodbox");

    colorInput.addEventListener("input", function() {
        moodbox.style.backgroundColor = colorInput.value;
    });
</script>

</body> </html>

Question:
Build a mini 'Now Playing' display like Spotify: show a song title in a <span id="song-title"></span> and a button 'Change Song'; when the button is clicked, update the innerHTML of the span to a new song name from a predefined array.

Answer:

<!DOCTYPE html>

<html lang="en"> <head> <meta charset="UTF-8"> <meta name="viewport" content="width=device-width, initial-scale=1.0"> <title>Now Playing</title> </head> <body>

<h2>Now Playing</h2>

<span id="song-title">Perfect</span>

<br><br>

<button id="changeSong">Change Song</button>

<script>
    let songs = ["Perfect", "Kesariya", "Believer", "Love Story"];
    let index = 0;

    document.getElementById("changeSong").addEventListener("click", function() {
        index++;

        if (index >= songs.length) {
            index = 0;
        }

        document.getElementById("song-title").innerHTML = songs[index];
    });
</script>

</body> </html>
