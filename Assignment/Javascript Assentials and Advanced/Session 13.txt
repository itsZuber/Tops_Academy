Question:
Create a function validateSpotifyURL(url) that takes a string and checks if it starts with 'https://open.spotify.com/'. If not, throw a custom Error with the message 'Invalid Spotify URL'.

Answer:

function validateSpotifyURL(url) {
    if (!url.startsWith("https://open.spotify.com/")) {
        throw new Error("Invalid Spotify URL");
    }

    return "Valid Spotify URL";
}

console.log(validateSpotifyURL("https://open.spotify.com/playlist/123"));

Question:
Create a function validateUsername(username) that checks if the username contains only letters and numbers; if not, use throw to raise a custom Error with the message 'Invalid Username'. Wrap the function call in try...catch and display the error message in the console.

Answer:

function validateUsername(username) {
    if (!/^[a-zA-Z0-9]+$/.test(username)) {
        throw new Error("Invalid Username");
    }

    return "Valid Username";
}

try {
    console.log(validateUsername("Zuber123"));
} catch (e) {
    console.log(e.message);
}

Question:
Write a function getOrderTotal(items) for a Zomato-style food order, where items is an array of objects with name and price. If any price is negative, throw a custom Error('Invalid item price'). Use try...catch to handle the error and display a friendly message.

Answer:

function getOrderTotal(items) {
    let total = 0;

    for (let item of items) {
        if (item.price < 0) {
            throw new Error("Invalid item price");
        }

        total = total + item.price;
    }

    return total;
}

let items = [
    {
        name: "Pizza",
        price: 300
    },
    {
        name: "Burger",
        price: -100
    },
    {
        name: "Cold Drink",
        price: 80
    }
];

try {
    console.log("Total Price: " + getOrderTotal(items));
} catch (e) {
    console.log("Sorry, there was a problem with your order.");
    console.log(e.message);
}

Question:
You are given this buggy code:

function formatFollowers(count) {
 try {
   if(count < 0) throw 'Negative count';
   return count + ' followers';
 } catch(e) {
   return e.message;
 }
}

It should return the error message if count is negative, but it returns undefined. Fix the code so it properly throws and catches an Error object.

Answer:

function formatFollowers(count) {
    try {
        if (count < 0) {
            throw new Error("Negative count");
        }

        return count + " followers";
    } catch (e) {
        return e.message;
    }
}

console.log(formatFollowers(-10));
console.log(formatFollowers(500));
