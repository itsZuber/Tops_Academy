Question:
Write a function addToCart(productName, quantity) that logs a success message using console.log(), a warning if quantity is zero using console.warn(), and an error if quantity is negative using console.error(). Call the function with all three scenarios to see each message type in the console.

Answer:

function addToCart(productName, quantity) {
    if (quantity < 0) {
        console.error("Quantity cannot be negative!");
    }
    else if (quantity == 0) {
        console.warn("Quantity is zero!");
    }
    else {
        console.log(productName + " added to cart. Quantity: " + quantity);
    }
}

addToCart("Laptop", 2);

addToCart("Mouse", 0);

addToCart("Keyboard", -1);

Write a JavaScript function addToCart(item, quantity) that accidentally throws an error when quantity is 0. Set a breakpoint in the Sources tab before the error line, run the function with quantity 0, and inspect the values of item and quantity at the breakpoint.

Answer:

function addToCart(item, quantity) {
    console.log("Item: " + item);
    console.log("Quantity: " + quantity);

    if (quantity == 0) {
        throw new Error("Quantity cannot be zero");
    }

    console.log(item + " added to cart.");
}

addToCart("Pizza", 0);

Question:

function getUser() {
    return fetchUser();
}

function fetchUser() {
    return JSON.parse(undefined);
}

getUser();

Answer:

function getUser() {
    return fetchUser();
}

function fetchUser() {
    return JSON.parse(undefined);
}

getUser();

The error occurs in the fetchUser() function at the line:

return JSON.parse(undefined);

The error happens because JSON.parse() cannot parse undefined as valid JSON.

The call stack is:

getUser()
    ↓
fetchUser()
    ↓
JSON.parse(undefined)

The exact line number depends on where the code is placed in the JavaScript file.
