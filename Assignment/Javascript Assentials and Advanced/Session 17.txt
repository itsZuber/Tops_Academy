Question:
Create a list of song cards (each with a title and an 'X' icon) using plain JavaScript and HTML. When you click the 'X' on any card, remove only that card from the DOM using event delegation (attach the click listener to the parent container, not each card).

Answer:

<!DOCTYPE html>
<html>
<head>
    <title>Song List</title>
</head>
<body>

    <h2>My Songs</h2>

    <div id="songList"></div>

    <script>
        let songs = [
            "Love Story",
            "Perfect",
            "Believer",
            "Night Changes"
        ];

        let songList = document.getElementById("songList");

        for (let song of songs) {
            let card = document.createElement("div");

            card.innerHTML = song + " <button class='remove'>X</button>";

            card.style.margin = "10px";

            songList.appendChild(card);
        }

        songList.addEventListener("click", function(event) {
            if (event.target.classList.contains("remove")) {
                event.target.parentElement.remove();
            }
        });
    </script>

</body>
</html>

Question:
Add a feature to your cart UI so that when a new product card is added dynamically (for example, by clicking an 'Add Random Product' button), the 'X' remove functionality still works without adding any new event listeners.

Answer:

<!DOCTYPE html>
<html>
<head>
    <title>Shopping Cart</title>
</head>
<body>

    <h2>Shopping Cart</h2>

    <button id="addProduct">Add Random Product</button>

    <div id="cart">

        <div class="product">
            <span>Laptop - ₹50000</span>
            <button class="remove">X</button>
        </div>

        <div class="product">
            <span>Headphones - ₹2000</span>
            <button class="remove">X</button>
        </div>

    </div>

    <script>
        let products = [
            "Smart Watch - ₹2500",
            "Keyboard - ₹1500",
            "Wireless Mouse - ₹800",
            "Mobile Phone - ₹20000"
        ];

        let cart = document.getElementById("cart");

        document.getElementById("addProduct").addEventListener("click", function() {
            let randomIndex = Math.floor(Math.random() * products.length);

            let product = document.createElement("div");

            product.className = "product";

            product.innerHTML =
                "<span>" + products[randomIndex] + "</span> " +
                "<button class='remove'>X</button>";

            cart.appendChild(product);
        });

        cart.addEventListener("click", function(event) {
            if (event.target.classList.contains("remove")) {
                event.target.parentElement.remove();
            }
        });
    </script>

</body>
</html>

Question:
Simulate a notification feed like YouTube, where each notification card has a 'Dismiss' button. When you click 'Dismiss', fade out and remove the card. Make sure the event delegation works for dynamically loaded notifications as well as those present at page load.

Answer:

<!DOCTYPE html>
<html>
<head>
    <title>Notifications</title>

    <style>
        .notification {
            padding: 10px;
            margin: 10px;
            border: 1px solid black;
            transition: opacity 0.5s;
        }

        .hide {
            opacity: 0;
        }
    </style>
</head>
<body>

    <h2>YouTube Notifications</h2>

    <button id="addNotification">Add Notification</button>

    <div id="notifications">

        <div class="notification">
            <span>New video uploaded!</span>
            <button class="dismiss">Dismiss</button>
        </div>

        <div class="notification">
            <span>Your channel has a new subscriber.</span>
            <button class="dismiss">Dismiss</button>
        </div>

    </div>

    <script>
        let notifications = document.getElementById("notifications");

        document.getElementById("addNotification").addEventListener("click", function() {
            let notification = document.createElement("div");

            notification.className = "notification";

            notification.innerHTML =
                "<span>New notification received!</span> " +
                "<button class='dismiss'>Dismiss</button>";

            notifications.appendChild(notification);
        });

        notifications.addEventListener("click", function(event) {
            if (event.target.classList.contains("dismiss")) {
                let card = event.target.parentElement;

                card.classList.add("hide");

                setTimeout(function() {
                    card.remove();
                }, 500);
            }
        });
    </script>

</body>
</html>

Question:
Given the following code snippet, fix the bug so that clicking the 'X' icon on any card removes only that card, even if cards are added dynamically:

document.querySelectorAll('.remove-btn').forEach(btn => {
    btn.addEventListener('click', function() {
        this.parentElement.remove();
    });
});

Answer:

The problem is that the event listeners are added only to the buttons that exist when the page loads. For dynamically added cards, the new buttons will not have a listener, so event delegation can be used by attaching one listener to the parent container.

<!DOCTYPE html>
<html>
<head>
    <title>Remove Cards</title>
</head>
<body>

    <div id="cards">

        <div class="card">
            <span>Song 1</span>
            <button class="remove-btn">X</button>
        </div>

        <div class="card">
            <span>Song 2</span>
            <button class="remove-btn">X</button>
        </div>

        <div class="card">
            <span>Song 3</span>
            <button class="remove-btn">X</button>
        </div>

    </div>

    <script>
        let cards = document.getElementById("cards");

        cards.addEventListener("click", function(event) {
            if (event.target.classList.contains("remove-btn")) {
                event.target.parentElement.remove();
            }
        });
    </script>

</body>
</html>
