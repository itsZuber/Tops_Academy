Given the following buggy code meant to calculate the total price of a Zomato order, identify and fix the syntax and runtime errors.

let prices = [120, 250, 90];

    // Check if the number is divisible by 2

Question 3:
Write a function formatFollowersCount(count) that takes a number and returns a formatted string like Instagram: 1500 as '1.5K', 1200000 as '1.2M', and numbers below 1000 as-is. Add clear comments and use proper indentation.

Answer:

// Format follower counts into K or M
function formatFollowersCount(count) {

    // Convert millions into M
    if (count >= 1000000) {
        return (count / 1000000).toFixed(1) + "M";
    }

    // Convert thousands into K
    else if (count >= 1000) {
        return (count / 1000).toFixed(1) + "K";
    }

    // Return numbers below 1000 as they are
    else {
        return count.toString();
    }
}

console.log(formatFollowersCount(1500));
console.log(formatFollowersCount(1200000));
console.log(formatFollowersCount(850));

// Output:
// 1.5K
// 1.2M
// 850


Question 4:
You are given a piece of JavaScript code that should print all even numbers from 1 to 10, but it doesn't work as expected. Find and fix the error, then rewrite the code with good indentation and at least one comment explaining the logic.

Answer:

for (let i = 1; i <= 10; i++) {

    // Check whether the number is even
    if (i % 2 == 0) {
        console.log(i);
    }
}

// Output:
// 2
// 4
// 6
// 8
