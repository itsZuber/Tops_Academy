Question 3:
Build a function that takes a 1D array of 7 integers representing your daily Zomato order amounts and calculates the average spend for the week.

Answer:

function calculateAverageSpend(orderAmounts) {
    let total = 0;

    for (let i = 0; i < orderAmounts.length; i++) {
        total = total + orderAmounts[i];
    }

    return total / orderAmounts.length;
}

let dailyOrders = [200, 350, 150, 400, 250, 300, 350];

console.log("Average weekly spend: " + calculateAverageSpend(dailyOrders));


Question 4:
Given a 2D array called cricketScores where each row represents an IPL match and columns represent runs scored by each team, write code to print the highest score from each match.

Answer:

let cricketScores = [
    [180, 165],
    [145, 190],
    [210, 205],
    [175, 185]
];

for (let i = 0; i < cricketScores.length; i++) {
    let highest = cricketScores[i][0];

    for (let j = 1; j < cricketScores[i].length; j++) {
        if (cricketScores[i][j] > highest) {
            highest = cricketScores[i][j];
        }
    }

    console.log("Match " + (i + 1) + " highest score: " + highest);
}
