Question 1:
Declare an integer variable called likes and a pointer variable called ptrLikes; assign likes a value, point ptrLikes to likes, and print both the value and the address stored in ptrLikes.

Answer:

#include <stdio.h>

int main() {
    int likes = 100;
    int *ptrLikes = &likes;

    printf("Likes value: %d\n", likes);
    printf("Address stored in ptrLikes: %p", (void*)ptrLikes);

    return 0;
}


Question 2:
Write a function swapPlaylistCounts(int *a, int *b) that swaps the number of songs in two Spotify playlists using pointers, then call the function in main and print the swapped values.

Answer:

#include <stdio.h>

void swapPlaylistCounts(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main() {
    int playlist1 = 25;
    int playlist2 = 40;

    swapPlaylistCounts(&playlist1, &playlist2);

    printf("Playlist 1 songs: %d\n", playlist1);
    printf("Playlist 2 songs: %d", playlist2);

    return 0;
}


Question 3:
Given an array of 5 order amounts (e.g., Zomato orders), use a pointer to iterate through the array and print each amount along with its memory address. Use pointer arithmetic to move to the next element.

Answer:

#include <stdio.h>

int main() {
    int orders[5] = {200, 350, 150, 400, 250};
    int *ptr = orders;
    int i;

    for (i = 0; i < 5; i++) {
        printf("Order amount: %d\n", *(ptr + i));
        printf("Memory address: %p\n", (void*)(ptr + i));
    }

    return 0;
}


Question 4:
Create a function incrementFollowers(int *followers, int n) that increases each follower count in an array (representing Instagram followers for 5 friends) by 100 using pointer arithmetic, then print the updated counts.

Answer:

#include <stdio.h>

void incrementFollowers(int *followers, int n) {
    int i;

    for (i = 0; i < n; i++) {
        *(followers + i) = *(followers + i) + 100;
    }
}

int main() {
    int followers[5] = {500, 800, 1000, 1200, 1500};
    int i;

    incrementFollowers(followers, 5);

    for (i = 0; i < 5; i++) {
        printf("%d\n", followers[i]);
    }

    return 0;
}
