SELECT * FROM Orders

Question 2:

INSERT INTO OrderItems (order_item_id, order_id, item_name, quantity)
VALUES (1, 101, 'Pizza', 1);

INSERT INTO OrderItems (order_item_id, order_id, item_name, quantity)
VALUES (2, 101, 'Burger', 2);

ROLLBACK;

SELECT * FROM OrderItems
WHERE order_id = 101;

No new items from these INSERT statements should remain after ROLLBACK, provided the transaction was not committed earlier.

Question 3:

INSERT INTO Cart (cart_id, user_id, product_name, quantity)
VALUES (1, 1, 'Laptop', 1);

SAVEPOINT product_added;

INSERT INTO Cart (cart_id, user_id, product_name, quantity)
VALUES (2, 1, 'Mouse', 1);

ROLLBACK TO product_added;

SELECT * FROM Cart
WHERE user_id = 1;

The Laptop remains in the cart, while the Mouse addition is undone, assuming the cart was initially empty for this user.

Question 4:

CREATE TABLE Wallet (
    user_id NUMBER PRIMARY KEY,
    balance NUMBER(10,2)
);

CREATE TABLE WalletTransactions (
    transaction_id NUMBER PRIMARY KEY,
    user_id NUMBER,
    amount NUMBER(10,2),
    transaction_type VARCHAR2(20)
);

CREATE OR REPLACE TRIGGER deduct_wallet_balance
AFTER INSERT ON WalletTransactions
FOR EACH ROW
BEGIN
    IF :NEW.transaction_type = 'EXPENSE' THEN
        UPDATE Wallet
        SET balance = balance - :NEW.amount
        WHERE user_id = :NEW.user_id;
    END IF;
END;
/
