    IF order_amount > 500 THEN
        DBMS_OUTPUT.PUT_LINE('Eligible for free delivery');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Delivery charges apply');

Question 3:

DECLARE
    counter NUMBER := 1;
BEGIN
    LOOP
        DBMS_OUTPUT.PUT_LINE(counter);
        counter := counter + 1;
        EXIT WHEN counter > 5;
    END LOOP;
END;
/

Question 4:

DECLARE
    CURSOR product_cursor IS
        SELECT PRODUCT_NAME FROM PRODUCTS;
    product_name PRODUCTS.PRODUCT_NAME%TYPE;
BEGIN
    OPEN product_cursor;
    LOOP
        FETCH product_cursor INTO product_name;
        EXIT WHEN product_cursor%NOTFOUND;
        DBMS_OUTPUT.PUT_LINE(product_name);
    END LOOP;
    CLOSE product_cursor;
END;
/

Question 5:

DECLARE
    num1 NUMBER := 10;
    num2 NUMBER := 0;
    result NUMBER;
BEGIN
    result := num1 / num2;
    DBMS_OUTPUT.PUT_LINE('Result: ' || result);
EXCEPTION
    WHEN ZERO_DIVIDE THEN
        DBMS_OUTPUT.PUT_LINE('Cannot divide by zero');
END;
/
