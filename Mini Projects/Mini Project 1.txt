Question 4:

#include <iostream>
#include <fstream>
using namespace std;

int main() {
    int minutes;
    int total = 0;
    int highest = 0;
    int days = 0;

    ifstream file("music_log.txt");

    while (file >> minutes) {
        total = total + minutes;

        if (minutes > highest) {
            highest = minutes;
        }

        days++;
    }

    file.close();

    if (days > 0) {
        cout << "Total minutes: " << total << endl;
        cout << "Average minutes: "
             << (double)total / days << endl;
        cout << "Highest minutes: " << highest << endl;
    }
    else {
        cout << "No saved data found.";
    }

    return 0;
}

#include <iostream>
#include <fstream>
using namespace std;

int main() {
    int minutes[7] = {0};
    int choice;
    char confirm;

    do {
        cout << "\n1. Log Listening Minutes";
        cout << "\n2. Weekly Report";
        cout << "\n3. Reset Weekly Data";
        cout << "\n4. Exit";
        cout << "\nEnter choice: ";
        cin >> choice;

        if (choice == 1) {
            ofstream file("music_log.txt");

            for (int i = 0; i < 7; i++) {
                cout << "Enter minutes for day " << i + 1 << ": ";
                cin >> minutes[i];

                file << minutes[i] << endl;
            }

            file.close();
            cout << "Data saved successfully.";
        }
        else if (choice == 2) {
            ifstream file("music_log.txt");

            int value;
            int total = 0;
            int highest = 0;
            int days = 0;

            while (file >> value) {
                total = total + value;

                if (value > highest) {
                    highest = value;
                }

                days++;
            }

            file.close();

            if (days > 0) {
                cout << "Total: " << total << " minutes" << endl;
                cout << "Average: "
                     << (double)total / days << " minutes" << endl;
                cout << "Highest: " << highest << " minutes" << endl;
            }
            else {
                cout << "No saved data found.";
            }
        }
        else if (choice == 3) {
            cout << "Are you sure you want to reset data? (Y/N): ";
            cin >> confirm;

            if (confirm == 'Y' || confirm == 'y') {
                for (int i = 0; i < 7; i++) {
                    minutes[i] = 0;
                }

                ofstream file("music_log.txt", ios::trunc);
                file.close();

                cout << "Weekly data reset successfully.";
            }
            else {
                cout << "Reset cancelled.";
            }
        }
        else if (choice == 4) {
            cout << "Exiting app...";
        }
        else {
            cout << "Invalid choice!";
        }

    } while (choice != 4);

    return 0;
}
