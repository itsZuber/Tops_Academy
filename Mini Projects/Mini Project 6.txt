Question 1:

Create an HTML form called musicLinksForm that accepts Spotify, YouTube Music, and Apple Music profile URLs as input fields, and display the submitted links as styled preview cards below the form.

Answer:

<!DOCTYPE html>
<html>
<head>
    <title>Music Links</title>

    <style>
        body {
            font-family: Arial;
            margin: 30px;
        }

        form {
            width: 400px;
            margin-bottom: 30px;
        }

        input {
            width: 100%;
            padding: 10px;
            margin: 8px 0;
            box-sizing: border-box;
        }

        button {
            padding: 10px 20px;
            background-color: black;
            color: white;
            border: none;
            cursor: pointer;
        }

        .card {
            width: 300px;
            padding: 15px;
            margin: 10px 0;
            border: 1px solid #ccc;
            border-radius: 8px;
        }
    </style>
</head>

<body>

    <h1>My Music Links</h1>

    <form id="musicLinksForm">

        <input
            type="url"
            id="spotify"
            placeholder="Spotify Profile URL"
            required
        >

        <input
            type="url"
            id="youtube"
            placeholder="YouTube Music Profile URL"
            required
        >

        <input
            type="url"
            id="apple"
            placeholder="Apple Music Profile URL"
            required
        >

        <button type="submit">Add Links</button>

    </form>

    <div id="preview"></div>

    <script>
        document.getElementById("musicLinksForm").addEventListener("submit", function(event) {

            event.preventDefault();

            var spotify = document.getElementById("spotify").value;
            var youtube = document.getElementById("youtube").value;
            var apple = document.getElementById("apple").value;

            document.getElementById("preview").innerHTML =
                '<div class="card"><h3>Spotify</h3><p>' + spotify + '</p></div>' +
                '<div class="card"><h3>YouTube Music</h3><p>' + youtube + '</p></div>' +
                '<div class="card"><h3>Apple Music</h3><p>' + apple + '</p></div>';
        });
    </script>

</body>
</html>

Question 4:

After submitting the form, show a preview card for each link with the platform logo, the username/handle extracted from the URL, and a 'Visit' button that opens the link in a new tab.

Answer:

<!DOCTYPE html>
<html>
<head>
    <title>Social Link Preview</title>

    <style>
        body {
            font-family: Arial;
            margin: 30px;
        }

        input {
            width: 350px;
            padding: 10px;
            margin: 8px 0;
        }

        button {
            padding: 10px 20px;
            background-color: black;
            color: white;
            border: none;
            cursor: pointer;
        }

        .card {
            width: 300px;
            padding: 15px;
            margin: 15px 0;
            border-radius: 8px;
            border: 2px solid;
        }

        .instagram {
            border-color: #e1306c;
        }

        .linkedin {
            border-color: #0077b5;
        }

        .spotify {
            border-color: #1db954;
        }

        .youtube {
            border-color: #ff0000;
        }

        .card img {
            width: 40px;
            height: 40px;
        }

        .card a {
            display: inline-block;
            padding: 8px 15px;
            background-color: black;
            color: white;
            text-decoration: none;
            border-radius: 5px;
        }
    </style>
</head>

<body>

    <h1>Add Social Link</h1>

    <form id="socialForm">

        <input
            type="url"
            id="socialUrl"
            placeholder="Enter social profile URL"
            required
        >

        <button type="submit">
            Add Link
        </button>

    </form>

    <div id="preview"></div>

    <script>

        document.getElementById("socialForm").addEventListener("submit", function(event) {

            event.preventDefault();

            var url = document.getElementById("socialUrl").value;

            var platform = "";
            var logo = "";
            var className = "";

            if (url.includes("instagram.com")) {

                platform = "Instagram";
                logo = "instagram.png";
                className = "instagram";

            } else if (url.includes("linkedin.com")) {

                platform = "LinkedIn";
                logo = "linkedin.png";
                className = "linkedin";

            } else if (url.includes("spotify.com")) {

                platform = "Spotify";
                logo = "spotify.png";
                className = "spotify";

            } else if (url.includes("youtube.com")) {

                platform = "YouTube";
                logo = "youtube.png";
                className = "youtube";

            } else {

                platform = "Social Profile";
                logo = "social.png";
                className = "";
            }

            var parts = url.split("/");
            var username = parts[parts.length - 1];

            if (username == "") {
                username = parts[parts.length - 2];
            }

            document.getElementById("preview").innerHTML +=
                '<div class="card ' + className + '">' +
                '<img src="' + logo + '" alt="' + platform + '">' +
                '<h3>' + platform + '</h3>' +
                '<p>@' + username + '</p>' +
                '<a href="' + url + '" target="_blank">Visit</a>' +
                '</div>';

            document.getElementById("socialForm").reset();

        });

    </script>

</body>
</html>
